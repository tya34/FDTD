# Run after the corresponding Tube FDTD simulation has finished.
E = getresult("yz_side","E");

y = E.y*1e6;
z = E.z*1e6;

Ex = pinch(E.Ex);
Ey = pinch(E.Ey);
Ez = pinch(E.Ez);

Emag = sqrt(abs(Ex)^2 + abs(Ey)^2 + abs(Ez)^2);
Emag = pinch(Emag);

image(y, z, Emag, "y (um)", "z (um)", "|E| Tube yz_side");

setplot("colorbar min",0);
setplot("colorbar max",3);

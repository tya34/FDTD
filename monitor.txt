E = getresult("field_yz_center","E");

z = E.z*1e6;
y = E.y*1e6;

Ex = pinch(E.Ex);
Ey = pinch(E.Ey);
Ez = pinch(E.Ez);

Emag = sqrt(abs(Ex)^2 + abs(Ey)^2 + abs(Ez)^2);
Emag = pinch(Emag);

image(y, z, Emag, "y (um)", "z (um)", "|E| field_yz_center");

setplot("colorbar min",0);
setplot("colorbar max",3);

# Run after the corresponding Helix FDTD simulation has finished.
E = getresult("xz_reference","E");

x = E.x*1e6;
z = E.z*1e6;

Ex = pinch(E.Ex);
Ey = pinch(E.Ey);
Ez = pinch(E.Ez);

Emag = sqrt(abs(Ex)^2 + abs(Ey)^2 + abs(Ez)^2);
Emag = pinch(Emag);

image(x, z, Emag, "x (um)", "z (um)", "|E| Helix xz_reference");

setplot("colorbar min",0);
setplot("colorbar max",3);

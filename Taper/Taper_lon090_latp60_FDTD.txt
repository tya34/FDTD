############################################################
# Taper_lon090_latp60_FDTD.txt
# Full-angle FDTD model using the SEM-calibrated curled membrane.
# Unrolled length/width = 200/250 um; VO2 thickness = 500 nm.
# The fixed y=0 edge remains on SiO2. Height rises continuously
# from both long edges toward the center ridge, and the two free
# corners meet at the common highest apex without curling past it.
############################################################

deleteall;
clear;

um = 1e-6;
nm = 1e-9;
pi_val = 3.141592653589793;

film_material = "VO2 20";
substrate_material = "SiO2 (Glass) - Palik";

# Monochromatic source wavelength used by every angle script.
lambda0 = 5*um;

# Unrolled membrane dimensions.
pattern_L = 200*um;
pattern_W = 250*um;
film_t = 500*nm;
half_W = 0.5*pattern_W;
substrate_top_z = 0;

# Shape controls.
apex_height = 105*um;
apex_y = 45*um;
free_center_height = 105*um;
free_center_y = 65*um;
center_ridge_lift = 18*um;
center_forward_bulge = 75*um;
edge_forward_bulge = 60*um;
free_lobe_half_width = 2.5*um;
curl_growth_power = 1.25;
free_y_power = 1.25;
free_z_power = 1.10;

# Surface resolution.
Nx = 180;
Ny = 180;
Nxv = Nx+1;
Nyv = Ny+1;
Nsurf = Nxv*Nyv;

addstructuregroup;
set("name","Taper_W250_L200_geometry");

# Mid-surface coordinates. flat_x and flat_y are material coordinates
# on the unrolled 250 um x 200 um rectangular membrane.
px = matrix(Nsurf,1);
py = matrix(Nsurf,1);
pz = matrix(Nsurf,1);

for(ix=1:Nxv){
    flat_x = -half_W + (ix-1)/Nx*pattern_W;

    for(iy=1:Nyv){
        flat_y = (iy-1)/Ny*pattern_L;
        y_frac = flat_y/pattern_L;

        # Smoothstep gives zero curl-rate slope at both ends.
        smooth_y = y_frac^2*(3-2*y_frac);
        curl_fraction = smooth_y^curl_growth_power;
        width_frac = flat_x/half_W;
        side_weight = abs(width_frac);
        if(width_frac < 0){
            side_sign = -1;
        } else {
            if(width_frac > 0){
                side_sign = 1;
            } else {
                side_sign = 0;
            }
        }

        # Centerline moves forward and then curls back. Each long edge
        # follows a shorter forward bow and ends at apex_y, so the
        # meeting point stays in front of rather than on the fixed edge.
        center_y = free_center_y*curl_fraction
                 + center_forward_bulge*4*y_frac*(1-y_frac);
        center_z = 0.5*film_t
                 + (free_center_height-film_t)*curl_fraction
                 + center_ridge_lift*4*curl_fraction*(1-curl_fraction);
        edge_x_abs = half_W*(1-curl_fraction);
        edge_y = apex_y*curl_fraction
               + edge_forward_bulge*4*y_frac*(1-y_frac);
        edge_z = 0.5*film_t
               + (apex_height-film_t)*curl_fraction;

        # Ruled base patch between the shared centerline and each side.
        # Its x sign is fixed by side_sign, so the two halves cannot cross.
        # center_z >= edge_z makes height increase monotonically from
        # either side toward the center ridge.
        base_x = side_sign*side_weight*edge_x_abs;
        base_y = (1-side_weight)*center_y + side_weight*edge_y;
        base_z = (1-side_weight)*center_z + side_weight*edge_z;

        # Exact free-edge target. It runs from the forward center point
        # to the apex on each side and leaves only a narrow opening.
        free_x = side_sign*free_lobe_half_width*sin(pi_val*side_weight);
        free_y = apex_y
               + (free_center_y-apex_y)*(1-side_weight)^free_y_power;
        free_z = free_center_height-0.5*film_t
               + (apex_height-free_center_height)*side_weight^free_z_power;
        free_base_y = (1-side_weight)*free_center_y
                    + side_weight*apex_y;
        free_base_z = (1-side_weight)*(free_center_height-0.5*film_t)
                    + side_weight*(apex_height-0.5*film_t);

        id = (iy-1)*Nxv + ix;
        px(id) = base_x + curl_fraction*free_x;
        py(id) = base_y + curl_fraction*(free_y-free_base_y);
        pz(id) = base_z + curl_fraction*(free_z-free_base_z);
    }
}

# Offset the mid-surface by +/- film_t/2 along its local normal to make
# a closed, nominally 500 nm thick solid.
vtx = matrix(2*Nsurf,3);
for(ix=1:Nxv){
    for(iy=1:Nyv){
        id = (iy-1)*Nxv + ix;

        if(ix == 1){
            idx1 = id+1;
            idx0 = id;
        } else {
            if(ix == Nxv){
                idx1 = id;
                idx0 = id-1;
            } else {
                idx1 = id+1;
                idx0 = id-1;
            }
        }

        if(iy == 1){
            idy1 = id+Nxv;
            idy0 = id;
        } else {
            if(iy == Nyv){
                idy1 = id;
                idy0 = id-Nxv;
            } else {
                idy1 = id+Nxv;
                idy0 = id-Nxv;
            }
        }

        tx_x = px(idx1)-px(idx0);
        tx_y = py(idx1)-py(idx0);
        tx_z = pz(idx1)-pz(idx0);
        ty_x = px(idy1)-px(idy0);
        ty_y = py(idy1)-py(idy0);
        ty_z = pz(idy1)-pz(idy0);

        normal_x = tx_y*ty_z - tx_z*ty_y;
        normal_y = tx_z*ty_x - tx_x*ty_z;
        normal_z = tx_x*ty_y - tx_y*ty_x;
        normal_mag = sqrt(normal_x^2 + normal_y^2 + normal_z^2);
        normal_x = normal_x/normal_mag;
        normal_y = normal_y/normal_mag;
        normal_z = normal_z/normal_mag;

        # The analytic normal of the fixed edge is +z, ensuring that its
        # lower film face lies exactly on the substrate at z=0.
        if(iy == 1){
            normal_x = 0;
            normal_y = 0;
            normal_z = 1;
        }

        # Give the two coincident free corners the same thickness
        # direction, so both faces also meet at the apex.
        if(iy == Nyv){
            if(ix == 1){
                normal_x = 0;
                normal_y = 0;
                normal_z = 1;
            }
            if(ix == Nxv){
                normal_x = 0;
                normal_y = 0;
                normal_z = 1;
            }
        }

        vtx(id,1) = px(id) + 0.5*film_t*normal_x;
        vtx(id,2) = py(id) + 0.5*film_t*normal_y;
        vtx(id,3) = pz(id) + 0.5*film_t*normal_z;
        vtx(id+Nsurf,1) = px(id) - 0.5*film_t*normal_x;
        vtx(id+Nsurf,2) = py(id) - 0.5*film_t*normal_y;
        vtx(id+Nsurf,3) = pz(id) - 0.5*film_t*normal_z;
    }
}

# Triangulate the upper and lower skins and close all four boundary walls.
Nfacets = 4*Nx*Ny + 4*Nx + 4*Ny;
fct = matrix(4,1,Nfacets);
fi = 1;

for(ix=1:Nx){
    for(iy=1:Ny){
        id00 = (iy-1)*Nxv + ix;
        id10 = id00 + 1;
        id01 = id00 + Nxv;
        id11 = id01 + 1;
        b00 = id00 + Nsurf;
        b10 = id10 + Nsurf;
        b01 = id01 + Nsurf;
        b11 = id11 + Nsurf;

        fct(1,1,fi)=id00; fct(2,1,fi)=id10; fct(3,1,fi)=id11; fi=fi+1;
        fct(1,1,fi)=id00; fct(2,1,fi)=id11; fct(3,1,fi)=id01; fi=fi+1;
        fct(1,1,fi)=b00; fct(2,1,fi)=b11; fct(3,1,fi)=b10; fi=fi+1;
        fct(1,1,fi)=b00; fct(2,1,fi)=b01; fct(3,1,fi)=b11; fi=fi+1;
    }
}

# Fixed and free short-edge walls.
for(ix=1:Nx){
    id0 = ix;
    id1 = ix+1;
    b0 = id0 + Nsurf;
    b1 = id1 + Nsurf;
    fct(1,1,fi)=id0; fct(2,1,fi)=b1; fct(3,1,fi)=b0; fi=fi+1;
    fct(1,1,fi)=id0; fct(2,1,fi)=id1; fct(3,1,fi)=b1; fi=fi+1;

    id0 = Ny*Nxv + ix;
    id1 = id0 + 1;
    b0 = id0 + Nsurf;
    b1 = id1 + Nsurf;
    fct(1,1,fi)=id0; fct(2,1,fi)=b0; fct(3,1,fi)=b1; fi=fi+1;
    fct(1,1,fi)=id0; fct(2,1,fi)=b1; fct(3,1,fi)=id1; fi=fi+1;
}

# The two long-edge walls; both move inward and return toward y=0,
# and their terminal points meet above the fixed-edge midpoint.
for(iy=1:Ny){
    id0 = (iy-1)*Nxv + 1;
    id1 = id0 + Nxv;
    b0 = id0 + Nsurf;
    b1 = id1 + Nsurf;
    fct(1,1,fi)=id0; fct(2,1,fi)=b0; fct(3,1,fi)=b1; fi=fi+1;
    fct(1,1,fi)=id0; fct(2,1,fi)=b1; fct(3,1,fi)=id1; fi=fi+1;

    id0 = (iy-1)*Nxv + Nxv;
    id1 = id0 + Nxv;
    b0 = id0 + Nsurf;
    b1 = id1 + Nsurf;
    fct(1,1,fi)=id0; fct(2,1,fi)=id1; fct(3,1,fi)=b1; fi=fi+1;
    fct(1,1,fi)=id0; fct(2,1,fi)=b1; fct(3,1,fi)=b0; fi=fi+1;
}

addplanarsolid;
set("name","Taper_W250_L200_VO2");
set("vertices",vtx);
set("facets",fct);
set("material",film_material);
addtogroup("Taper_W250_L200_geometry");

# Finite visualization substrate. Its top surface is z=0 and the complete
# fixed edge of the membrane lies on it.
fdtd_margin_x = 50*um;
fdtd_margin_y = 50*um;
substrate_xy_overhang = 5*um;
substrate_margin_x = fdtd_margin_x + substrate_xy_overhang;
substrate_margin_y = fdtd_margin_y + substrate_xy_overhang;
substrate_depth = 10*um;

addrect;
set("name","SiO2_substrate");
set("x",0);
set("x span",pattern_W + 2*substrate_margin_x);
set("y",0.5*pattern_L);
set("y span",pattern_L + 2*substrate_margin_y);
set("z",substrate_top_z - 0.5*substrate_depth);
set("z span",substrate_depth);
set("material",substrate_material);
addtogroup("Taper_W250_L200_geometry");

# Geometry checks printed to the Lumerical script prompt.
left_tip_id = Ny*Nxv + 1;
right_tip_id = Ny*Nxv + Nxv;
tip_separation = sqrt((px(left_tip_id)-px(right_tip_id))^2 + (py(left_tip_id)-py(right_tip_id))^2 + (pz(left_tip_id)-pz(right_tip_id))^2);
tip_offset_from_fixed_midpoint = sqrt(px(left_tip_id)^2 + py(left_tip_id)^2);
?"Unrolled membrane L x W x t = "+num2str(pattern_L/um)+" x "+num2str(pattern_W/um)+" x "+num2str(film_t/nm)+" um, um, nm";
?"Free-corner separation = "+num2str(tip_separation/nm)+" nm";
?"Apex XY offset from fixed-edge midpoint = "+num2str(tip_offset_from_fixed_midpoint/nm)+" nm";

# Extract the actual curled-film bounds for the FDTD region, sources,
# and field-monitor locations used by every illumination angle.
film_min_x = vtx(1,1);
film_max_x = vtx(1,1);
film_min_y = vtx(1,2);
film_max_y = vtx(1,2);
film_min_z = vtx(1,3);
film_max_z = vtx(1,3);
for(vertex_index=2:2*Nsurf){
    if(vtx(vertex_index,1) < film_min_x){ film_min_x = vtx(vertex_index,1); }
    if(vtx(vertex_index,1) > film_max_x){ film_max_x = vtx(vertex_index,1); }
    if(vtx(vertex_index,2) < film_min_y){ film_min_y = vtx(vertex_index,2); }
    if(vtx(vertex_index,2) > film_max_y){ film_max_y = vtx(vertex_index,2); }
    if(vtx(vertex_index,3) < film_min_z){ film_min_z = vtx(vertex_index,3); }
    if(vtx(vertex_index,3) > film_max_z){ film_max_z = vtx(vertex_index,3); }
}

fdtd_substrate_depth = 4*um;
source_gap_above_taper = 8*um;
source_top_gap = 4*um;

fdtd_xmin = film_min_x - fdtd_margin_x;
fdtd_xmax = film_max_x + fdtd_margin_x;
fdtd_ymin = film_min_y - fdtd_margin_y;
fdtd_ymax = film_max_y + fdtd_margin_y;
fdtd_zmin = substrate_top_z - fdtd_substrate_depth;
source_z = film_max_z + source_gap_above_taper;
fdtd_zmax = source_z + source_top_gap;

fdtd_xc = (fdtd_xmin+fdtd_xmax)/2;
fdtd_yc = (fdtd_ymin+fdtd_ymax)/2;
fdtd_zc = (fdtd_zmin+fdtd_zmax)/2;
fdtd_xs = fdtd_xmax-fdtd_xmin;
fdtd_ys = fdtd_ymax-fdtd_ymin;
fdtd_zs = fdtd_zmax-fdtd_zmin;

monitor_zmin = substrate_top_z - 3*um;
monitor_zmax = source_z - 1*um;
monitor_zc = (monitor_zmin+monitor_zmax)/2;
monitor_zs = monitor_zmax-monitor_zmin;

addfdtd;
set("x",fdtd_xc);
set("x span",fdtd_xs);
set("y",fdtd_yc);
set("y span",fdtd_ys);
set("z",fdtd_zc);
set("z span",fdtd_zs);
set("mesh accuracy",1);
set("x min bc","PML");
set("x max bc","PML");
set("y min bc","PML");
set("y max bc","PML");
set("z min bc","PML");
set("z max bc","PML");

# User source coordinates: source_lon_deg = 90 deg, source_lat_deg = 60 deg.
# Earth-like source hemisphere: longitude runs from -x through +z to +x; latitude runs from -y to +y.
# Source-position unit vector: src_pos_x = 0, src_pos_y = 0.866025, src_pos_z = 0.5.
# Target propagation unit vector from the source position back toward the structure center:
# kx = 0, ky = -0.866025, kz = -0.5.
# User longitude/latitude are source-position coordinates; FDTD angle theta/phi below are Lumerical source angles.
# The injection axis is selected from the largest |k| component to avoid grazing source injection.
source_lon_deg = 90;
source_lat_deg = 60;
src_pos_x = 0;
src_pos_y = 0.866025;
src_pos_z = 0.5;
target_kx = 0;
target_ky = -0.866025;
target_kz = -0.5;
fdtd_angle_theta_deg = 30;
fdtd_angle_phi_deg = 0;

addplane;
set("name","src_lon090_latp60");
set("plane wave type","Bloch/Periodic");
set("injection axis","y");
set("direction","backward");
set("angle theta",fdtd_angle_theta_deg);
set("angle phi",fdtd_angle_phi_deg);
set("polarization angle",90);
set("x",fdtd_xc);
set("x span",fdtd_xs);
set("y",fdtd_ymax - 0.5*source_top_gap);
set("z",fdtd_zc);
set("z span",fdtd_zs);
set("wavelength start",lambda0);
set("wavelength stop",lambda0);

addprofile; set("name","field_xy_lower"); set("monitor type",7);
set("x",fdtd_xc); set("x span",0.9*fdtd_xs); set("y",fdtd_yc); set("y span",0.9*fdtd_ys); set("z",film_min_z + 0.35*(film_max_z-film_min_z));
set("override global monitor settings",1); set("use source limits",1); set("frequency points",1);

addprofile; set("name","field_xy_upper"); set("monitor type",7);
set("x",fdtd_xc); set("x span",0.9*fdtd_xs); set("y",fdtd_yc); set("y span",0.9*fdtd_ys); set("z",film_min_z + 0.75*(film_max_z-film_min_z));
set("override global monitor settings",1); set("use source limits",1); set("frequency points",1);

addprofile; set("name","field_yz_center"); set("monitor type",5);
set("x",fdtd_xc); set("y",fdtd_yc); set("y span",0.9*fdtd_ys); set("z",monitor_zc); set("z span",monitor_zs);
set("override global monitor settings",1); set("use source limits",1); set("frequency points",1);

addprofile; set("name","field_yz_side"); set("monitor type",5);
set("x",fdtd_xc + 0.25*(film_max_x-film_min_x)); set("y",fdtd_yc); set("y span",0.9*fdtd_ys); set("z",monitor_zc); set("z span",monitor_zs);
set("override global monitor settings",1); set("use source limits",1); set("frequency points",1);

addprofile; set("name","field_xz_reference"); set("monitor type",6);
set("x",fdtd_xc); set("x span",0.9*fdtd_xs); set("y",fdtd_yc); set("z",monitor_zc); set("z span",monitor_zs);
set("override global monitor settings",1); set("use source limits",1); set("frequency points",1);

# run;

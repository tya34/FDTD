# Run after the corresponding 3V FDTD simulation has finished.
E = getresult("xy_upper","E");

x = E.x*1e6;
y = E.y*1e6;

Ex = pinch(E.Ex);
Ey = pinch(E.Ey);
Ez = pinch(E.Ez);

Emag = sqrt(abs(Ex)^2 + abs(Ey)^2 + abs(Ez)^2);
Emag = pinch(Emag);

image(x, y, Emag, "x (um)", "y (um)", "|E| 3V xy_upper");

setplot("colorbar min",0);
setplot("colorbar max",3);
